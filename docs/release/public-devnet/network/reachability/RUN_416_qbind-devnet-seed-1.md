## 11. Exact command(s) used / verifier command(s)

```bash
# SEED (VPS, host ubuntu-s-1vcpu-1gb-sgp1) — externally reachable KEMTLS static-root listener
qbind-node --env devnet --network-mode p2p --enable-p2p \
    --p2p-listen-addr 0.0.0.0:30333 \
    --p2p-advertised-addr 188.166.227.87:30333 \
    --p2p-mutual-auth required --p2p-pqc-root-mode pqc-static-root \
    --p2p-trusted-root <temp-devnet-static-root> \
    --p2p-leaf-cert <temp-leaf.cert.bin> --p2p-leaf-cert-key <temp-leaf.kem.sk.bin>

# OFF-HOST DIALER (Laptop 1 WSL, host olivegigi) — independent, different network
qbind-node --env devnet --network-mode p2p --enable-p2p \
    --p2p-mutual-auth required --p2p-pqc-root-mode pqc-static-root \
    --p2p-trusted-root <temp-devnet-static-root> \
    --p2p-leaf-cert <temp-leaf.cert.bin> --p2p-leaf-cert-key <temp-leaf.kem.sk.bin> \
    --validator-id <override> \
    --p2p-peer 0@188.166.227.87:30333        # per-peer KEM public key override supplied

nc -vz 188.166.227.87 30333

# temporary DevNet PQC material (both sides; ephemeral; discarded)
cargo run --release --example devnet_pqc_root_helper -- <outdir> <num_validators>
```

material was discarded after the run; no key/cert bytes are committed.
